cmake_minimum_required(VERSION 3.25)
project(FailureLab VERSION 1.0.1 LANGUAGES CXX)
include(CTest)
include(FetchContent)
option(FAILURELAB_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(FAILURELAB_TIDY "Run clang-tidy during compilation" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(json)
add_library(failurelab_core src/config.cpp src/policy.cpp src/simulation.cpp src/report.cpp)
target_include_directories(failurelab_core PUBLIC include)
target_link_libraries(failurelab_core PUBLIC nlohmann_json::nlohmann_json)
target_compile_definitions(failurelab_core PUBLIC FAILURELAB_VERSION="${PROJECT_VERSION}")
add_library(failurelab_options INTERFACE)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(failurelab_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
  if(FAILURELAB_SANITIZERS)
    target_compile_options(failurelab_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(failurelab_options INTERFACE -fsanitize=address,undefined)
  endif()
elseif(FAILURELAB_SANITIZERS)
  message(FATAL_ERROR "Sanitizers require GCC or Clang")
endif()
if(FAILURELAB_TIDY)
  find_program(TIDY_EXECUTABLE NAMES clang-tidy clang-tidy-18 REQUIRED)
  execute_process(COMMAND "${TIDY_EXECUTABLE}" --version OUTPUT_VARIABLE TIDY_VERSION RESULT_VARIABLE TIDY_STATUS)
  if(NOT TIDY_STATUS EQUAL 0 OR NOT TIDY_VERSION MATCHES "version 18[.]1[.]8([^0-9]|$)")
    message(FATAL_ERROR "clang-tidy 18.1.8 is required; set TIDY_EXECUTABLE to the pinned executable")
  endif()
  set_target_properties(failurelab_core PROPERTIES CXX_CLANG_TIDY "${TIDY_EXECUTABLE};--warnings-as-errors=*")
endif()
target_link_libraries(failurelab_core PUBLIC failurelab_options)
add_executable(failurelab src/main.cpp)
target_link_libraries(failurelab PRIVATE failurelab_core)
if(FAILURELAB_TIDY)
  set_target_properties(failurelab PROPERTIES CXX_CLANG_TIDY "${TIDY_EXECUTABLE};--warnings-as-errors=*")
endif()
install(TARGETS failurelab RUNTIME DESTINATION bin)
install(DIRECTORY scenarios/ DESTINATION share/failurelab/scenarios)
install(FILES third_party/nlohmann-json.LICENSE.MIT DESTINATION share/failurelab/licenses)
if(BUILD_TESTING)
  set(DOCTEST_NO_INSTALL ON CACHE BOOL "Do not install test framework" FORCE)
  FetchContent_Declare(doctest URL https://github.com/doctest/doctest/archive/refs/tags/v2.4.11.tar.gz URL_HASH SHA256=632ed2c05a7f53fa961381497bf8069093f0d6628c5f26286161fbd32a560186 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(doctest)
  add_executable(failurelab_tests tests/unit.cpp)
  target_link_libraries(failurelab_tests PRIVATE failurelab_core doctest::doctest)
  add_test(NAME unit COMMAND failurelab_tests)
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:failurelab> ${CMAKE_CURRENT_SOURCE_DIR})
  if(APPLE)
    set(LEAK_CHECK 0) # Apple ASan does not support LeakSanitizer.
  else()
    set(LEAK_CHECK 1)
  endif()
  set_tests_properties(unit integration PROPERTIES ENVIRONMENT "ASAN_OPTIONS=detect_leaks=${LEAK_CHECK}:halt_on_error=1;UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
endif()

# Local binary archives: no publishing/uploading is performed by this target.
install(FILES README.md THIRD_PARTY.md CHANGELOG.md DESTINATION share/doc/failurelab)
install(DIRECTORY docs/ DESTINATION share/doc/failurelab/docs)
install(DIRECTORY examples/ DESTINATION share/doc/failurelab/examples)
install(FILES docs/package-readme.txt DESTINATION . RENAME README.txt)
set(CPACK_GENERATOR TGZ)
set(CPACK_PACKAGE_NAME FailureLab)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME "FailureLab-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_BINARY_DIR}/packages")
include(CPack)
